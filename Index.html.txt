  <script>
    // Platzhalter – hier können die echten Verse eingefügt werden
    const atVerses = Array.from({length: 365}, (_, i) => `AT Vers ${i+1} — Elberfelder 1905`);
    const ntVerses = Array.from({length: 365}, (_, i) => `NT Vers ${i+1} — Elberfelder 1905`);

    function getDayIndex() {
      const start = new Date(new Date().getFullYear(), 0, 1);
      const now = new Date();
      return Math.floor((now - start) / (1000*60*60*24)) % atVerses.length;
    }

    document.getElementById("at").innerText = "AT: " + atVerses[getDayIndex()];
    document.getElementById("nt").innerText = "NT: " + ntVerses[getDayIndex()];
  </script>
